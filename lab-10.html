3.4 Equidistance boundary from Mexico and Canada

cities <- cities %>%
  mutate(dist_to_mexico = set_units(st_distance(., mexico_combine), "km")) %>% 
  mutate(dist_to_canada = set_units(st_distance(., canada_combine), "km"))
eqdist <- cities %>% 
  mutate(equidist_diff = as.numeric(dist_to_mexico) - as.numeric(dist_to_canada))

eqdist_zone <- eqdist %>%
  mutate(abs_diff = abs(equidist_diff)) %>%
  slice_min(abs_diff, n = 20)

top_5_eqdist <- eqdist_zone %>%
